2. 提取分组列表（在加载数据的 useEffect 中）：

```tsx
useEffect(() => {
  if (isOpen) {
    chrome.storage.local.get(["records"], (result) => {
      const data = result.records || {};
      const allRecords = Object.values(data) as TabRecord[];
      setRecords(allRecords);

      // 提取唯一分组
      const uniqueGroups = Array.from(
        new Map(
          allRecords.map((r) => [
            r.groupName,
            { name: r.groupName, color: r.groupColor },
          ]),
        ).values(),
      );
      setGroups(uniqueGroups);

      setSearchResults(allRecords);
    });
  }
}, [isOpen]);
```

3. 修改搜索逻辑，支持分组过滤：

```tsx
useEffect(() => {
  let filtered = records;

  // 先按分组过滤
  if (selectedGroup) {
    filtered = filtered.filter((r) => r.groupName === selectedGroup);
  }

  // 再按关键词搜索
  if (!searchQuery.trim()) {
    setSearchResults(filtered);
    return;
  }

  const timer = setTimeout(() => {
    const fuse = new Fuse(filtered, {
      keys: ["title", "url", "groupName"],
      threshold: 0.3,
      ignoreLocation: true,
    });

    const results = fuse.search(searchQuery);
    setSearchResults(results.map((r) => r.item));
  }, 300);

  return () => clearTimeout(timer);
}, [searchQuery, records, selectedGroup]);
```

4. 在搜索框下方添加分组标签按钮：

```tsx
{
  /* 分组过滤标签 */
}
{
  groups.length > 0 && (
    <div className="px-4 py-2 border-b flex gap-2 flex-wrap">
      <button
        onClick={() => setSelectedGroup(null)}
        className={`px-3 py-1 text-sm rounded ${
          selectedGroup === null
            ? "bg-blue-500 text-white"
            : "bg-gray-100 text-gray-700 hover:bg-gray-200"
        }`}
      >
        全部
      </button>
      {groups.map((group) => (
        <button
          key={group.name}
          onClick={() => setSelectedGroup(group.name)}
          className={`px-3 py-1 text-sm rounded ${
            selectedGroup === group.name ? "ring-2 ring-blue-500" : ""
          }`}
          style={{
            backgroundColor: group.color || "#gray",
            color: "#fff",
          }}
        >
          {group.name}
        </button>
      ))}
    </div>
  );
}
```
